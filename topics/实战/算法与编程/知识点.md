## 链表

### 反转链表

```js
function reverseList(head) {
  let prev = null;
  let curr = head;
  while (curr) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}
```

### 找链表中间节点（快慢指针）

```js
function findMiddle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
  }
  return slow;
}
```

## 树

### 二叉树遍历

```js
// 前序（根→左→右）— 递归
function preorder(root) {
  if (!root) return [];
  return [root.val, ...preorder(root.left), ...preorder(root.right)];
}

// 前序 — 迭代（用栈）
function preorderIterative(root) {
  if (!root) return [];
  const result = [], stack = [root];
  while (stack.length) {
    const node = stack.pop();
    result.push(node.val);
    if (node.right) stack.push(node.right); // 右子先入栈
    if (node.left) stack.push(node.left);   // 左子后入栈（先出）
  }
  return result;
}

// 层序（逐层遍历）— BFS
function levelOrder(root) {
  if (!root) return [];
  const result = [], queue = [root];
  while (queue.length) {
    const level = [];
    const size = queue.length;
    for (let i = 0; i < size; i++) {
      const node = queue.shift();
      level.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
    result.push(level);
  }
  return result;
}
```

### 从前序和中序遍历还原二叉树

```js
function buildTree(preorder, inorder) {
  if (!preorder.length) return null;
  const rootVal = preorder[0];
  const mid = inorder.indexOf(rootVal);
  return {
    val: rootVal,
    left: buildTree(preorder.slice(1, mid + 1), inorder.slice(0, mid)),
    right: buildTree(preorder.slice(mid + 1), inorder.slice(mid + 1))
  };
}
```

## 字符串

### 最长无重复子串（滑动窗口）

```js
function lengthOfLongestSubstring(s) {
  const set = new Set();
  let left = 0, max = 0;
  for (let right = 0; right < s.length; right++) {
    while (set.has(s[right])) {
      set.delete(s[left++]);
    }
    set.add(s[right]);
    max = Math.max(max, right - left + 1);
  }
  return max;
}
```

```js
function canPermutePalindrome(s) {
  const count = {};
  for (const c of s) count[c] = (count[c] || 0) + 1;
  return Object.values(count).filter(v => v % 2 === 1).length <= 1;
}
```

## 数组

### 数组扁平化

```js
// 递归
function flatten(arr) {
  return arr.reduce((acc, cur) =>
    acc.concat(Array.isArray(cur) ? flatten(cur) : cur), []);
}

// ES2019 原生
arr.flat(Infinity);

// 迭代（用栈，避免栈溢出）
function flattenIterative(arr) {
  const stack = [...arr].reverse();
  const result = [];
  while (stack.length) {
    const item = stack.pop();
    Array.isArray(item) ? stack.push(...item.reverse()) : result.push(item);
  }
  return result;
}
```

### 洗牌算法（Fisher-Yates）

```js
function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}
```

**为什么 `Array.sort(() => Math.random() - 0.5)` 不公平？**
- sort 的比较次数不是均匀的，每个元素被比较的次数不同
- 导致某些排列出现的概率远高于其他排列
- Fisher-Yates 每个排列概率完全相等（O(n) 时间复杂度）

### V8 sort 实现

- ≤10 个元素：**插入排序**
- >10 个元素：**TimSort**（归并排序变种）
- TimSort 先找有序段（run），对每个 run 做插入排序，再归并合并

### 快速排序

```js
function quickSort(arr) {
  if (arr.length <= 1) return arr;
  const pivot = arr[Math.floor(arr.length / 2)];
  const left = [], right = [], equal = [];
  for (const x of arr) {
    if (x < pivot) left.push(x);
    else if (x > pivot) right.push(x);
    else equal.push(x);
  }
  return [...quickSort(left), ...equal, ...quickSort(right)];
}
```

## 拓扑排序

**场景：** 任务调度、模块加载顺序、课程先修关系

```js
function topoSort(nodes, edges) {
  const inDegree = new Map();
  const adj = new Map();
  nodes.forEach(n => { inDegree.set(n, 0); adj.set(n, []); });
  edges.forEach(([from, to]) => {
    adj.get(from).push(to);
    inDegree.set(to, inDegree.get(to) + 1);
  });

  const queue = nodes.filter(n => inDegree.get(n) === 0);
  const result = [];
  while (queue.length) {
    const node = queue.shift();
    result.push(node);
    for (const next of adj.get(node)) {
      inDegree.set(next, inDegree.get(next) - 1);
      if (inDegree.get(next) === 0) queue.push(next);
    }
  }
  return result.length === nodes.length ? result : null; // null 表示有环
}
```

```js
function climbStairs(n) {
  if (n <= 2) return n;
  let a = 1, b = 2;
  for (let i = 3; i <= n; i++) {
    [a, b] = [b, a + b];
  }
  return b;
}
```

### 全排列

```js
function permutations(str) {
  const result = [];
  function backtrack(path, remaining) {
    if (remaining.length === 0) { result.push(path); return; }
    for (let i = 0; i < remaining.length; i++) {
      backtrack(path + remaining[i], remaining.slice(0, i) + remaining.slice(i + 1));
    }
  }
  backtrack('', str);
  return result;
}
```

```js
function cartesian(...sets) {
  return sets.reduce((acc, set) =>

}
```

### Q: 拖拽功能怎么实现？

```js
element.addEventListener('mousedown', (e) => {
  const startX = e.clientX - element.offsetLeft;
  const startY = e.clientY - element.offsetTop;

  const onMove = (e) => {
    element.style.left = `${e.clientX - startX}px`;
    element.style.top = `${e.clientY - startY}px`;
  };

  const onUp = () => {
    document.removeEventListener('mousemove', onMove);
    document.removeEventListener('mouseup', onUp);
  };

  document.addEventListener('mousemove', onMove);
  document.addEventListener('mouseup', onUp);
});
```
