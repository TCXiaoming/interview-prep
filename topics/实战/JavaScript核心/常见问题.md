## Q1: 实现一个 once 函数，确保传入的函数只被执行一次

```js
function once(fn) {
  let called = false;
  let result;
  return function(...args) {
    if (!called) {
      called = true;
      result = fn.apply(this, args);
    }
    return result;
  };
}

// 使用
const init = once(() => {
  console.log('init');
  return 'done';
});
init(); // 'init', 'done'
init(); // 'done'（不打印）
```

## Q3: 手动实现一个简易的响应式数据系统

```js
class Reactive {
  constructor(data) {
    this.data = data;
    this.deps = new Map(); // key → [effectFn]
    this.activeEffect = null;
    this.observe(data);
  }

  observe(obj) {
    Object.keys(obj).forEach(key => {
      let value = obj[key];
      const self = this;
      Object.defineProperty(obj, key, {
        get() {
          if (self.activeEffect) {
            if (!self.deps.has(key)) self.deps.set(key, []);
            self.deps.get(key).push(self.activeEffect);
          }
          return value;
        },
        set(newVal) {
          value = newVal;
          // 触发依赖
          (self.deps.get(key) || []).forEach(fn => fn());
        }
      });
    });
  }

  // 注册副作用
  effect(fn) {
    this.activeEffect = fn;
    fn(); // 执行时触发 get，完成依赖收集
    this.activeEffect = null;
  }
}

// 使用
const r = new Reactive({ count: 0, name: 'test' });
r.effect(() => console.log('count changed:', r.data.count));
r.data.count = 1; // 自动触发 effect
```

## Q4: 实现一个图片上传管理器，始终保持 N 张在上传

```js
class UploadManager {
  constructor(maxConcurrency) {
    this.max = maxConcurrency;
    this.running = 0;
    this.queue = [];
  }

  add(file) {
    this.queue.push(file);
    this.run();
  }

  run() {
    while (this.running < this.max && this.queue.length > 0) {
      const file = this.queue.shift();
      this.running++;
      this.upload(file).finally(() => {
        this.running--;
        // 每完成一张，自动补上新的
        this.run();
      });
    }
  }

  async upload(file) {
    const formData = new FormData();
    formData.append('file', file);
    const res = await fetch('/api/upload', { method: 'POST', body: formData });
    // 上传完成后通知外部补一张新的
    this.onComplete?.(file, await res.json());
  }
}
```

## Q5: 请描述 DOM 的事件模型

| 级别 | 特点 |
|---|---|
| DOM Level 0 | `onclick` 属性，同一事件只能绑定一个处理函数 |
| DOM Level 2 | `addEventListener`，支持冒泡/捕获阶段，可绑定多个处理函数 |
| DOM Level 3 | 增加更多事件类型（`DOMContentLoaded`、`hashchange`、`touch` 等） |

**事件流三个阶段：**
1. **捕获阶段** — 从 window → document → html → body → 目标元素
2. **目标阶段** — 到达目标元素
3. **冒泡阶段** — 从目标元素向上冒泡

`stopPropagation()` 阻止传播，`preventDefault()` 阻止默认行为。

## Q6: V8 中隐藏类（Hidden Class）对对象创建效率的影响

**参考回答：**

V8 用隐藏类来优化对象属性访问。相同结构的对象共享同一个隐藏类。

```js
// 高效：所有对象属性顺序一致，共享隐藏类
function Point(x, y) { this.x = x; this.y = y; }
const p1 = new Point(1, 2);
const p2 = new Point(3, 4);
// p1 和 p2 共享同一个隐藏类

// 低效：动态添加属性，每个对象的隐藏类不同
const p1 = {};
p1.x = 1;
p1.y = 2;
const p2 = {};
p2.y = 4; // 属性添加顺序不同
p2.x = 3;
// p1 和 p2 的隐藏类不同，V8 无法做内联缓存优化
```

结论：构造函数创建对象比对象字面量动态添加属性更高效（在大量创建时有感知）。

## Q7: 数组的 sort() 在 V8 中如何实现？

V8 对短数组（≤10 个元素）用**插入排序**，对长数组用**TimSort**（归并排序的变种，源自 Python）。

TimSort 的特点：
- 先找到数据中的"天然有序段"（run）
- 对每个 run 做插入排序（小规模数据插入排序更快）
- 然后用归并排序合并这些 run
- 最坏时间复杂度 O(n log n)，最好 O(n)（数据已经基本有序时）

## Q8: 判断循环引用

```js
function hasCircularReference(obj, seen = new WeakSet()) {
  if (obj === null || typeof obj !== 'object') return false;
  if (seen.has(obj)) return true;
  seen.add(obj);
  return Object.values(obj).some(val => hasCircularReference(val, seen));
}
```

## Q9: Event loop 执行顺序题

```js
console.log('1'); // 同步

setTimeout(() => {
  console.log('2'); // 宏任务
  Promise.resolve().then(() => console.log('3')); // 微任务（在宏任务内）
}, 0);

Promise.resolve().then(() => {
  console.log('4'); // 微任务
  setTimeout(() => console.log('5'), 0); // 宏任务
});

console.log('6'); // 同步

// 输出顺序：1 → 6 → 4 → 2 → 3 → 5
```

## Q10: bind 链式调用的 this 问题

```js
const obj = { name: 'obj' };
function fn() { return this.name; }

const bound1 = fn.bind(obj);
const bound2 = bound1.bind({ name: 'other' });

bound2(); // 'obj' — bind 只有第一次生效
// 原因：bind 返回的函数的 this 已经被锁定，后续 bind 无法改变
```
